\begin{frame}{The question}
\begin{columns}[c]
\column{0.53\textwidth}
\large
Can optimal-transport flow matching replace a CycleGAN for \textbf{unpaired} jet translation: clean
PYTHIA (vacuum) $\to$ JEWEL-like (medium)?

\vspace{0.8em}
\normalsize
Two things must hold:
\begin{enumerate}
\item the outputs look like real JEWEL jets, statistically;
\item each output still depends on its own input jet.
\end{enumerate}

\vspace{0.6em}
\smallnote{There is no true JEWEL partner for a PYTHIA jet: any unpaired map is one of many with the
same distributions. So this tests distributions and dependence, not a per-jet medium modification.}
\column{0.47\textwidth}
\centering

\begin{frame}{Two ways to learn an unpaired map}
\begin{columns}[T]
\column{0.5\textwidth}
\keyot\ot{} \refs{1, 2}
\begin{itemize}\setlength\itemsep{0.3em}
\item pair each batch of 256 PYTHIA and 256 JEWEL jets by optimal transport (least change in log energy)
\item regress the velocity along straight lines between the pairs: a plain mean-squared-error loss, no
  adversary
\item translate: solve the flow ODE, 128 network calls
\end{itemize}
\column{0.5\textwidth}
\keycg\cg{} \refs{3--5}: the repository's UVCGAN2, which UVCGAN-S builds on
\begin{itemize}\setlength\itemsep{0.3em}
\item two generators (PYTHIA $\to$ JEWEL, JEWEL $\to$ PYTHIA) and two discriminators
\item adversarial, cycle-consistency and identity losses
\item translate: one pass of the PYTHIA $\to$ JEWEL generator
\end{itemize}
\end{columns}
\vspace{1.2em}
\colorbox{wash}{\parbox{0.97\linewidth}{\vspace{0.4em}\hspace{0.3em}\textbf{Held equal:} the same 200k + 200k
training jets (the leading R = 0.4 jet, 53 towers on a 16 $\times$ 16 grid), the same log-energy
coordinates, the same generator network (the UVCGAN-S ViT-ModNet, 32M parameters), seed 0, and
\textbf{2 hours on one A6000}. CycleGAN also continues to 8 and 24 hours.\vspace{0.4em}}}
\end{frame}

\begin{frame}{Caveat: the $p_T$ change follows the two samples}
\begin{columns}[c]
\column{0.5\textwidth}
\includegraphics[width=\textwidth]{deck_ptchange.png}
\column{0.5\textwidth}
\begin{itemize}\setlength\itemsep{0.4em}
\item the PYTHIA sample starts near 20 GeV (presumably its generator-level jet requirement); JEWEL
  reaches down to 10 GeV
\item \ot{} lowers 20-25 GeV jets by 40\% and leaves jets above 40 GeV unchanged: a map between two
  differently selected samples, not a quenching mechanism
\item \cg{} at 2 hours raises $p_T$ by a factor growing from 1.2 (15-20 GeV inputs) to 2.6 (above 50
  GeV): not a map between the two spectra
\end{itemize}
\end{columns}
\end{frame}

% ---------------------------------------------------------------- 10
\begin{frame}{Summary}
\begin{itemize}\setlength\itemsep{0.5em}
\item \textbf{At the same 2 GPU hours:} \ot{} (no pairs) produces jets statistically indistinguishable from
  held-out JEWEL ($p_T$, 7 substructure observables, their $p_T$ dependence, re-found jets), and each output
  stays tied to its input
\item \cg{} (the repository's UVCGAN2, same data and generator network) does not: its $p_T$ scale is off
  by a factor 2 and no observable moves toward JEWEL; its outputs do keep their inputs
\item its training had not settled at 2 hours (the 1-hour checkpoint erred the other way); its 8- and
  24-hour checkpoints are being trained, and this deck is rebuilt with them
\item \ot{}'s $p_T$ change reflects how the two samples were selected, not quenching; no per-jet truth
  exists
\end{itemize}
\end{frame}
